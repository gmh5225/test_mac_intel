cmake_minimum_required(VERSION 3.30)
cmake_policy(SET CMP0091 NEW)
project(owned_windows_ugrep_sources LANGUAGES C CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION OFF)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(ZLIB_BUILD_EXAMPLES ON CACHE BOOL "" FORCE)
add_subdirectory("${SOURCE_ROOT}/zlib" zlib)
add_library(owned_bzip2 STATIC
  "${SOURCE_ROOT}/bzip2/blocksort.c" "${SOURCE_ROOT}/bzip2/huffman.c"
  "${SOURCE_ROOT}/bzip2/crctable.c" "${SOURCE_ROOT}/bzip2/randtable.c"
  "${SOURCE_ROOT}/bzip2/compress.c" "${SOURCE_ROOT}/bzip2/decompress.c" "${SOURCE_ROOT}/bzip2/bzlib.c")
set(ZSTD_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(ZSTD_USE_STATIC_RUNTIME ON CACHE BOOL "" FORCE)
set(ZSTD_BUILD_STATIC ON CACHE BOOL "" FORCE)
set(ZSTD_BUILD_PROGRAMS ON CACHE BOOL "" FORCE)
set(ZSTD_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ZSTD_ZLIB_SUPPORT OFF CACHE BOOL "" FORCE)
set(ZSTD_LZMA_SUPPORT OFF CACHE BOOL "" FORCE)
set(ZSTD_LZ4_SUPPORT OFF CACHE BOOL "" FORCE)
add_subdirectory("${SOURCE_ROOT}/zstd/build/cmake" zstd)
set(BROTLI_DISABLE_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory("${SOURCE_ROOT}/brotli" brotli)
set(PCRE2_BUILD_PCRE2_8 ON CACHE BOOL "" FORCE)
set(PCRE2_BUILD_PCRE2_16 OFF CACHE BOOL "" FORCE)
set(PCRE2_BUILD_PCRE2_32 OFF CACHE BOOL "" FORCE)
set(PCRE2_BUILD_PCRE2GREP ON CACHE BOOL "" FORCE)
set(PCRE2_BUILD_TESTS ON CACHE BOOL "" FORCE)
set(PCRE2_SUPPORT_JIT ON CACHE BOOL "" FORCE)
set(PCRE2_STATIC_RUNTIME ON CACHE BOOL "" FORCE)
add_subdirectory("${SOURCE_ROOT}/pcre2" pcre2)

set(UG "${SOURCE_ROOT}/ugrep")
set(VIIZ 7zAlloc 7zArcIn 7zBuf 7zBuf2 7zCrc 7zCrcOpt 7zDec 7zFile 7zStream
  Bcj2 Bra Bra86 BraIA64 CpuArch Delta Lzma2Dec LzmaDec Ppmd7 Ppmd7Dec viizip)
list(TRANSFORM VIIZ PREPEND "${UG}/lzma/C/")
list(TRANSFORM VIIZ APPEND ".c")
add_library(owned_viiz STATIC ${VIIZ})
target_compile_definitions(owned_viiz PRIVATE Z7_PPMD_SUPPORT Z7_EXTRACT_ONLY NDEBUG)

set(REFLEX convert debug error input matcher matcher_avx2 matcher_avx512bw pattern posix
  simd simd_avx2 simd_avx512bw unicode utf8 block_scripts language_scripts letter_scripts letter_case composer)
list(TRANSFORM REFLEX PREPEND "${UG}/lib/")
list(TRANSFORM REFLEX APPEND ".cpp")
set(UGCPP ugrep cnf glob output query screen stats vkey)
list(TRANSFORM UGCPP PREPEND "${UG}/src/")
list(TRANSFORM UGCPP APPEND ".cpp")
add_library(owned_ugrep_source OBJECT ${REFLEX} ${UGCPP} "${UG}/src/zopen.c")
target_include_directories(owned_ugrep_source PRIVATE
  "${UG}/include" "${UG}/lzma/C" "${SOURCE_ROOT}/pcre2/src" "${CMAKE_CURRENT_BINARY_DIR}/pcre2"
  "${SOURCE_ROOT}/zlib" "${CMAKE_CURRENT_BINARY_DIR}/zlib" "${SOURCE_ROOT}/bzip2"
  "${SOURCE_ROOT}/zstd/lib" "${SOURCE_ROOT}/brotli/c/include")
target_compile_options(owned_ugrep_source PRIVATE "/FI${UG_CONFIG}")
target_compile_definitions(owned_ugrep_source PRIVATE
  _CRT_SECURE_NO_WARNINGS _CRT_NONSTDC_NO_WARNINGS WITH_COLOR WITH_NO_INDENT WITH_NO_CODEGEN
  HAVE_PCRE2 HAVE_LIBZ HAVE_LIBBZ2 HAVE_LIBZSTD HAVE_LIBBROTLI PCRE2_STATIC)
if(RECOVERY_ARCH STREQUAL "arm64")
  target_compile_definitions(owned_ugrep_source PRIVATE HAVE_NEON)
elseif(RECOVERY_ARCH STREQUAL "x64")
  target_compile_definitions(owned_ugrep_source PRIVATE HAVE_SSE2)
else()
  message(FATAL_ERROR "Only the measured native Windows targets are supported")
endif()
add_executable(ugrep-source-host $<TARGET_OBJECTS:owned_ugrep_source>)
target_link_libraries(ugrep-source-host PRIVATE owned_viiz pcre2-8-static zlibstatic owned_bzip2 libzstd_static brotlidec brotlicommon)
set(RECOVERY_LIBRARY_PATHS "")
foreach(target IN ITEMS owned_viiz pcre2-8-static zlibstatic owned_bzip2 libzstd_static brotlidec brotlicommon)
  string(APPEND RECOVERY_LIBRARY_PATHS "${target}\t$<TARGET_FILE:${target}>\n")
endforeach()
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/recovery-library-paths.tsv" CONTENT "${RECOVERY_LIBRARY_PATHS}")
enable_testing()
