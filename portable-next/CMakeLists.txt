cmake_minimum_required(VERSION 3.22)
project(RecordProbe LANGUAGES C CXX)
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
endif()
add_subdirectory("${CAPSTONE_SOURCE}" capstone)
foreach(linkage static shared)
    add_executable(probe_${linkage} probe.cpp)
    target_compile_features(probe_${linkage} PRIVATE cxx_std_17)
    target_link_libraries(probe_${linkage} PRIVATE capstone_${linkage} ${CMAKE_DL_LIBS})
    if(APPLE)
        set_target_properties(probe_${linkage} PROPERTIES BUILD_WITH_INSTALL_RPATH ON INSTALL_RPATH "@loader_path")
    elseif(UNIX)
        set_target_properties(probe_${linkage} PROPERTIES BUILD_WITH_INSTALL_RPATH ON INSTALL_RPATH "$ORIGIN")
    endif()
endforeach()
